# 1. An admin token and a user token (scripts/generate-jwt.sh <user-id> [role])
USER_ID=00000000-0000-0000-0000-000000000001
TOKEN=$(./scripts/generate-jwt.sh $USER_ID 2>/dev/null)
./scripts/generate-jwt.sh $USER_ID 2>/dev/null
ADMIN=$(./scripts/generate-jwt.sh $USER_ID admin 2>/dev/null)
./scripts/generate-jwt.sh $USER_ID admin 2>/dev/null
# 2. A wallet, funded through the front door this time — no hand-posting to the ledger
WALLET=$(curl -s -X POST http://localhost:8080/wallets -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" -d '{"currency":"BRL"}' | jq -r .walletId)
curl -s -X POST http://localhost:8080/wallets -H "Authorization: Bearer $TOKEN"   -H "Content-Type: application/json" -d '{"currency":"BRL"}' | jq -r .walletId
curl -s -X POST http://localhost:8080/deposits \
  -H "Authorization: Bearer $ADMIN" -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d "{\"destinationWalletId\":\"$WALLET\",\"amount\":\"500.00\"}" | jq

curl -s -o /dev/null -w '%{http_code}\n' -X POST http://localhost:8080/deposits \

  -d "{\"destinationWalletId\":\"$WALLET\",\"amount\":\"1.00\"}"      # -> 403
403
# 4. Lifecycle: freeze blocks transfers, close refuses a funded wallet
curl -s -X POST http://localhost:8080/wallets/$WALLET/freeze   -H "Authorization: Bearer $TOKEN" -i | head -1
HTTP/1.1 200 
curl -s -X POST http://localhost:8080/wallets/$WALLET/unfreeze -H "Authorization: Bearer $TOKEN" -i | head -1
HTTP/1.1 200

# 6. THE MONEY SHOT — break a projection on purpose, watch it get caught, then fix it
#    First wait for the projection to show the step-2 deposit: on a cold stack the consumer is
#    still joining its group, and corrupting a row that does not exist yet is a silent UPDATE 0.
until curl -s http://localhost:8080/wallets/$WALLET/balance -H "Authorization: Bearer $TOKEN"   | jq -e '.balance == 500' >/dev/null; do sleep 1; done
docker exec ledger-platform-postgres-1 psql -U ledger -d ledger -c \
  "UPDATE projection_db.wallet_balances SET balance = 999999.00 WHERE wallet_id = '$WALLET';"
UPDATE 1
